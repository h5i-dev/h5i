05 / Formal verification

Prove correctness with Lean 4

Rust / kernel/src/lib.rs
fn delete(user: u64, s: &Snapshot,
          id: u64) -> Outcome {
    match find_post(&s.posts, id) {
        None => Err(Error::NotFound),
        Some(p) => {
            if p.author == user
                || is_moderator(&s.moderators, user) {
                Ok((one(Write::DelPost(id)),
                    Reply::Done))
            } else {
                Err(Error::Forbidden)
            }
        }
    }
}
