Rust / parallel/kernel/src/lib.rs · pure kernel
pub fn push_rule(event: &[u8], pattern: &[u8],
    display: &[u8], members: u64,
    minimum: u64, maximum: u64, power: u64,
    required: u64, enabled: bool,
    require_display: bool) -> u32 {
    if !enabled { return 0; }
    if power < required { return 0; }
    if members < minimum
        || members > maximum { return 0; }
    if !glob(event, pattern) { return 0; }
    // display-name check omitted
    1
}

05 / Verify

Verify application logic with Lean 4

Rust / kernel/src/lib.rs
fn delete(user: u64, s: &Snapshot,
          id: u64) -> Outcome {
    match find_post(&s.posts, id) {
        None => Err(Error::NotFound),
        Some(p) => {
            if p.author == user
                || is_moderator(&s.moderators, user) {
                Ok((one(Write::DelPost(id)),
                    Reply::Done))
            } else {
                Err(Error::Forbidden)
            }
        }
    }
}
